> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task.

**Goal:** 将云同步从「API Key 自托管」重构为「官方账户登录」：默认免登录使用核心功能，注册/登录后自动启用云同步。

**Architecture:** 后端重构为账户体系（users/tokens/records 三张表）+ Bearer token 鉴权 + KEK 环境变量加密存储；客户端新增 auth-client.js，改造 sync-client.js 用 token 替换 API Key，设置面板替换为登录/登出 UI。

**Tech Stack:** Node 内置 crypto/http/fs/path（后端零依赖）；CommonJS + globalThis 双导出（同 history.js）。

```js
const crypto = require("crypto");
const ALGO = "aes-256-gcm";
const IV_LEN = 16;
const TAG_LEN = 16;

function randomHex() {
  return crypto.randomBytes(32).toString("hex");
}

let _kek = null;
function loadKek() {
  if (_kek) return _kek;
  const hex = process.env.SYNC_KEK;
  if (!hex) throw new Error("SYNC_KEK env is required (64 hex chars = 32 bytes)");
  _kek = Buffer.from(hex, "hex");
  if (_kek.length !== 32) throw new Error("SYNC_KEK must be 64 hex chars (AES-256)");
  return _kek;
}

function encrypt(plaintext) {
  const kek = loadKek();
  const iv = crypto.randomBytes(IV_LEN);
  const cipher = crypto.createCipheriv(ALGO, kek, iv);
  const ct = Buffer.concat([cipher.update(plaintext, "utf8"), cipher.final()]);
  const tag = cipher.getAuthTag();
  return {
    ciphertext: Buffer.concat([ct, tag]).toString("base64"),
    iv: iv.toString("base64"),
  };
}

function decrypt(ciphertext, iv) {
  const kek = loadKek();
  const raw = Buffer.from(ciphertext, "base64");
  const ct = raw.slice(0, raw.length - TAG_LEN);
  const tag = raw.slice(raw.length - TAG_LEN);
  const decipher = crypto.createDecipheriv(ALGO, kek, Buffer.from(iv, "base64"));
  decipher.setAuthTag(tag);
  return Buffer.concat([decipher.update(ct), decipher.final()]).toString("utf8");
}

module.exports = { randomHex, loadKek, encrypt, decrypt, ALGO };
```

```js
const fs = require("fs");
const path = require("path");

function createStore(dataDir) {
  fs.mkdirSync(dataDir, { recursive: true });
  const file = path.join(dataDir, "store.json");
  let data = { users: {}, tokens: {}, records: {} };
  if (fs.existsSync(file)) {
    try { data = JSON.parse(fs.readFileSync(file, "utf8")); } catch (_) { data = { users: {}, tokens: {}, records: {} }; }
  }
  const save = () => fs.writeFileSync(file, JSON.stringify(data));

  return {
    // users
    getUser(email) { return data.users[email] || null; },
    hasUser(email) { return !!data.users[email]; },
    putUser(email, rec) { data.users[email] = rec; save(); },
    // tokens
    putToken(email) {
      const token = require("crypto").randomBytes(32).toString("hex");
      data.tokens[token] = { email, createdAt: Date.now() };
      save();
      return token;
    },
    getTokenUser(token) { return (data.tokens[token] && data.tokens[token].email) || null; },
    revokeToken(token) { if (data.tokens[token]) { delete data.tokens[token]; save(); } },
    // records (key: "email:id")
    _key(email, id) { return email + ":" + id; },
    getRecord(email, id) { return data.records[this._key(email, id)] || null; },
    putRecord(email, id, rec) { data.records[this._key(email, id)] = rec; save(); },
    delRecord(email, id) { if (data.records[this._key(email, id)]) { delete data.records[this._key(email, id)]; save(); } },
    listRecords(email) {
      return Object.keys(data.records)
        .filter((k) => k.startsWith(email + ":"))
        .map((k) => data.records[k]);
    },
    save,
  };
}

module.exports = { createStore };
```

```js
const crypto = require("crypto");

function hashPassword(password, salt) {
  return crypto.scryptSync(password, salt, 64).toString("hex");
}

function makePassword() {
  return crypto.randomBytes(16).toString("hex");
}

function verifyPassword(password, salt, expectedHash) {
  const a = Buffer.from(hashPassword(password, salt), "hex");
  const b = Buffer.from(expectedHash, "hex");
  return a.length === b.length && crypto.timingSafeEqual(a, b);
}

module.exports = { hashPassword, makePassword, verifyPassword };
```

```js
const { createStore } = require("./store.js");
const { hashPassword, makePassword, verifyPassword } = require("./auth.js");
const { encrypt, decrypt } = require("./crypto.js");

function createApp(opts) {
  const store = opts && opts.store || createStore(opts && opts.dataDir || "data");
  const kek = require("./crypto.js").loadKek();

  return (req, res) => {
    const send = (status, obj) => {
      res.statusCode = status;
      res.setHeader("Content-Type", "application/json");
      res.end(JSON.stringify(obj));
    };
    const readBody = () => new Promise((resolve) => {
      let body = "";
      req.on("data", (c) => body += c);
      req.on("end", () => { try { resolve(JSON.parse(body || "{}")); } catch (_) { resolve(null); } });
    });
    const bearer = (req.headers["authorization"] || "").replace(/^Bearer\s+/i, "");
    const authedEmail = bearer && store.getTokenUser(bearer);

    const url = new URL(req.url, "http://localhost");

    if (url.pathname === "/v1/health" && req.method === "GET") {
      return send(200, { ok: true });
    }

    // auth endpoints (no bearer required)
    if (url.pathname === "/v1/auth/register" && req.method === "POST") {
      readBody().then((body) => {
        if (!body || !body.email || !body.password) return send(400, { error: "email and password required" });
        if (store.hasUser(body.email)) return send(409, { error: "email already registered" });
        const salt = makePassword();
        store.putUser(body.email, { email: body.email, passwordHash: hashPassword(body.password, salt), salt, createdAt: Date.now() });
        const token = store.putToken(body.email);
        send(200, { token, email: body.email });
      });
      return;
    }

    if (url.pathname === "/v1/auth/login" && req.method === "POST") {
      readBody().then((body) => {
        if (!body || !body.email || !body.password) return send(400, { error: "email and password required" });
        const user = store.getUser(body.email);
        if (!user || !verifyPassword(body.password, user.salt, user.passwordHash)) return send(401, { error: "invalid credentials" });
        const token = store.putToken(body.email);
        send(200, { token, email: body.email });
      });
      return;
    }

    if (url.pathname === "/v1/auth/logout" && req.method === "POST") {
      if (bearer) store.revokeToken(bearer);
      return send(200, { ok: true });
    }

    if (url.pathname === "/v1/auth/me" && req.method === "GET") {
      if (!authedEmail) return send(401, { error: "unauthorized" });
      return send(200, { email: authedEmail });
    }

    if (!url.pathname.startsWith("/v1/")) return send(404, { error: "not found" });
    if (!authedEmail) return send(401, { error: "unauthorized" });

    if (url.pathname === "/v1/records" && req.method === "GET") {
      const recs = store.listRecords(authedEmail).map((r) => {
        const plain = decrypt(r.ciphertext, r.iv);
        return { ...JSON.parse(plain), _email: undefined };
      });
      return send(200, recs);
    }

    const m = url.pathname.match(/^\/v1\/records\/([^/]+)$/);
    if (!m) return send(404, { error: "not found" });
    const id = decodeURIComponent(m[1]);

    if (req.method === "PUT") {
      readBody().then((body) => {
        if (!body || !body.id) return send(400, { error: "bad record" });
        const plaintext = JSON.stringify({ ...body, _email: authedEmail });
        const enc = encrypt(plaintext);
        store.putRecord(authedEmail, id, { id, ...enc });
        send(200, { ok: true });
      });
      return;
    }

    if (req.method === "DELETE") {
      store.delRecord(authedEmail, id);
      return send(200, { ok: true });
    }

    send(405, { error: "method not allowed" });
  };
}

if (require.main === module) {
  const port = Number(process.env.PORT) || 8181;
  const dataDir = process.env.DATA_DIR || "data";
  require("http").createServer(createApp({ dataDir })).listen(port, () => {
    console.log("mio-auth-server on :" + port);
  });
}

module.exports = { createApp };
```

```js
const SESSION_KEY = "mioSession";

async function register(email, password, serverUrl) {
  const r = await fetch(serverUrl.replace(/\/+$/, "") + "/v1/auth/register", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ email, password }),
  });
  if (!r.ok) { const e = new Error("register failed"); e.status = r.status; throw e; }
  const data = await r.json();
  const session = { token: data.token, email: data.email };
  await chrome.storage.local.set({ [SESSION_KEY]: session });
  return session;
}

async function login(email, password, serverUrl) {
  const r = await fetch(serverUrl.replace(/\/+$/, "") + "/v1/auth/login", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ email, password }),
  });
  if (!r.ok) { const e = new Error("login failed"); e.status = r.status; throw e; }
  const data = await r.json();
  const session = { token: data.token, email: data.email };
  await chrome.storage.local.set({ [SESSION_KEY]: session });
  return session;
}

async function logout(serverUrl) {
  try {
    await fetch(serverUrl.replace(/\/+$/, "") + "/v1/auth/logout", {
      method: "POST",
      headers: { "Authorization": "Bearer " + getSession().token },
    });
  } catch (_) {}
  await chrome.storage.local.remove(SESSION_KEY);
}

function getSession() {
  // 同步版本不可用（需 async）；调用方用 _getSessionAsync
  return _cached || null;
}
let _cached = null;

async function _getSessionAsync() {
  if (_cached) return _cached;
  const raw = await chrome.storage.local.get(SESSION_KEY);
  _cached = raw[SESSION_KEY] || null;
  return _cached;
}

async function isLoggedIn() {
  const s = await _getSessionAsync();
  return !!(s && s.token);
}

async function clearSession() {
  _cached = null;
  await chrome.storage.local.remove(SESSION_KEY);
}

if (typeof module !== "undefined") {
  module.exports = { register, login, logout, isLoggedIn, getSession: _getSessionAsync, clearSession, SESSION_KEY };
} else {
  globalThis.AuthClient = { register, login, logout, isLoggedIn, getSession: _getSessionAsync, clearSession, SESSION_KEY };
}
```

```js
  const headers = { "Authorization": "Bearer " + token };
  if (body) headers["Content-Type"] = "application/json";
  const resp = await fetch(url, { method: method || "GET", headers, body: body ? JSON.stringify(body) : undefined });
  if (!resp.ok) { const err = new Error("HTTP " + resp.status); err.status = resp.status; throw err; }
  return resp.status === 204 ? null : resp.json();
}

// listRemote/putRecord 签名加 token
async function listRemote(serverUrl, token) { return _req(serverUrl, token, "records", "GET"); }
async function putRecord(serverUrl, token, bundle) {
  return _req(serverUrl, token, "records/" + encodeURIComponent(bundle.id), "PUT", bundle);
}

// syncHistory 签名改为 (serverUrl, token, localList)
async function syncHistory(serverUrl, token, localList) {
  const remoteBundles = await listRemote(serverUrl, token);
  const remote = [];
  const failed = [];
  for (const b of remoteBundles) {
    try { remote.push(b); } catch (_) { failed.push(b.id); }
  }
  const { merged, pulled, toPush } = mergeRecords(localList, remote);
  let pushed = 0;
  for (const rec of toPush) {
    try { await putRecord(serverUrl, token, sanitizeRecord(rec)); pushed++; }
    catch (e) { failed.push(rec.id); }
  }
  return { merged, pulled: pulled.length, pushed, pushFailed: toPush.length - pushed, failed };
}
```
